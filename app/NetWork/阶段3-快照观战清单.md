## 3. 关键代码

### 3.1 ProtoCodec.cs

    using App.Protocol;      // 生成的 GameState / PlayerState / FoodState / Point
    using App.World;         // WorldSnapShot / PlayerSnapShot / FoodSnapShot

    namespace App.NetWork;

    public static class ProtoCodec
    {
        public static GameState ToProto(WorldSnapShot snap, string roomId)
        {
            var msg = new GameState
            {
                RoomId = roomId,
                ServerTick = snap.Tick,
                IsGameOver = snap.IsOver,
            };

            foreach (var p in snap.Players)
            {
                var ps = new PlayerState
                {
                    PlayerId = p.Id,
                    Score = p.Score,
                    IsAlive = p.IsAlive,
                };

                foreach (var seg in p.Segaments)
                    ps.SnakeSegments.Add(new Point { X = seg.X, Y = seg.Y });

                msg.Players.Add(ps);
            }

            foreach (var f in snap.Foods)
                msg.Foods.Add(new FoodState
                {
                    Position = new Point { X = f.Position.X, Y = f.Position.Y },
                    FoodType = f.Type,
                });

            return msg;
        }
    }

> 注意：`seg` 是 `App.Services.State.Point`，`new Point` 是 `App.Protocol.Point`，两个同名类型别混。

### 3.2 SnapshotPump.cs

    using App.World;
    using Google.Protobuf;          // ToByteArray()

    namespace App.NetWork;

    public sealed class SnapshotPump
    {
        readonly GameWorld world;
        readonly string roomId;
        readonly MqttTransport transport;
        readonly int intervalMs;
        readonly CancellationTokenSource cts = new();

        public SnapshotPump(GameWorld world, string roomId, MqttTransport transport, int intervalMs = 100)
        {
            this.world = world;
            this.roomId = roomId;
            this.transport = transport;
            this.intervalMs = intervalMs;
        }

        public void Start() => _ = RunAsync(cts.Token);

        async Task RunAsync(CancellationToken ct)
        {
            using var timer = new PeriodicTimer(TimeSpan.FromMilliseconds(intervalMs));
            while (await timer.WaitForNextTickAsync(ct))
            {
                if (world.Players.Count == 0) continue;      // 还没开局，别发

                var msg = ProtoCodec.ToProto(world.Snapshot(), roomId);
                byte[] bytes = msg.ToByteArray();
                if (!await transport.PublishAsync($"snake/{roomId}/state", bytes, ct))
                    transport.Log?.Invoke("[pump] 发布失败");
            }
        }
    }

> 关键点：pump **不挂在游戏循环里**，自己一个定时器读 world.Snapshot()，所以单人和双人逻辑一行不用改。

### 3.3 NetHost 里的 Attach（新增）

    /// <summary>游戏侧把世界交给网络层（GamePrepareArgs 构造末尾调用一次）</summary>
    public static void Attach(GamePrepareArgs args)
    {

    }

### 3.5 RemoteRender.cs（客户端画棋盘）

    using App.Config;
    using App.Protocol;
    using System.Text;

    namespace App.NetWork;

    public static class RemoteRender
    {
        static GameSettings? settings;

        public static void Draw(GameState state)
        {
            settings ??= GameSettingLoader.Load();
            int w = settings.MapConfig.GameMapWidth;
            int h = settings.MapConfig.GameMapHeight;

            var grid = new char[h, w];
            for (int y = 0; y < h; y++)
                for (int x = 0; x < w; x++)
                    grid[y, x] = (x == 0 || y == 0 || x == w - 1 || y == h - 1) ? '█' : ' ';

            foreach (var f in state.Foods)
                grid[f.Position.Y, f.Position.X] = '✰';

            for (int i = 0; i < state.Players.Count; i++)
            {
                var p = state.Players[i];
                foreach (var seg in p.SnakeSegments)
                    grid[seg.Y, seg.X] = p.IsAlive ? '◊' : '.';

                var head = p.SnakeSegments[^1];
                if (p.IsAlive)
                    grid[head.Y, head.X] = i == 0 ? '☻' : '☺';
            }

            var sb = new StringBuilder();
            for (int y = 0; y < h; y++)
            {
                for (int x = 0; x < w; x++) sb.Append(grid[y, x]);
                sb.AppendLine();
            }

            Console.Clear();
            Console.WriteLine($"————— 观战  房间={state.RoomId}  tick={state.ServerTick} —————");
            Console.Write(sb);

            foreach (var p in state.Players)
                Console.WriteLine($"{p.PlayerId}: {p.Score} 分  {(p.IsAlive ? "存活" : "死亡")}");

            if (state.IsGameOver) Console.WriteLine("[对局结束]");
        }
    }

### 3.6 NetHost 的 Host / Client 分支改造

    // Host：连上就返回，让 program.cs 继续进游戏（pump 在 Attach 里启动）
    static Task RunHostAsync()
    {
        Console.WriteLine("[host] 已就绪，进入游戏后开始广播快照");
        return Task.CompletedTask;
    }

    // Client：只收 state 并渲染
    static async Task RunClientAsync()
    {
        Console.WriteLine("[client] 等待主机快照…（按 q 退出）");
        using var timer = new PeriodicTimer(TimeSpan.FromMilliseconds(50));

        while (await timer.WaitForNextTickAsync())
        {
            while (mqttTransport!.TryReceive(out var topic, out var payload))
            {
                if (!topic.EndsWith("/state", StringComparison.Ordinal)) continue;
                RemoteRender.Draw(GameState.Parser.ParseFrom(payload));
            }

            while (Console.KeyAvailable)
                if (Console.ReadKey(true).Key is ConsoleKey.Q) return;
        }
    }

## 4. 字段映射（现有 proto 够不够）

| 运行时 WorldSnapShot | proto GameState | 说明 |
| --- | --- | --- |
| Tick | server_tick | 用于判断「这帧是不是新的」 |
| Players[].Id / Score / IsAlive | players[].player_id / score / is_alive | 够用 |
| Players[].Segaments | players[].snake_segments | 顺序：头在最后（Snake[^1]） |
| Foods[] | foods[] | 够用 |
| IsOver | is_game_over | 够用 |
| Width / Height / RemainingSeconds / Name / Direction | **proto 里没有** | 客户端从 Config/GameSettings.json 读宽高；名字/方向本期不显示 |

> 本阶段不改 app.proto，避免牵动生成代码。

## 7. 常见问题

| 现象 | 原因 | 处理 |
| --- | --- | --- |
| Client 一直白屏 | 没收到 state：topic 不匹配 | 检查订阅是不是 snake/{room}/#，房间里 state 与 room 是否一致 |
| Client 收到但解析崩 | 把 hello 的文本当 proto 解析了 | 只处理 /state 结尾的 topic |
| 画面闪烁严重 | 每 50ms Console.Clear | 降到 10Hz 或改用 SetCursorPosition(0,0) |
| Host 一开局就卡 | pump 在 UI 线程上跑 | pump 用了独立定时器，确认没有 await 在主循环里 |
| 快照发不出去 | PublishAsync 返回 false | 看 [mqtt] 日志，多半是连接已断 |
